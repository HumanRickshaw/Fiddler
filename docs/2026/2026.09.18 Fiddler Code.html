def getViridis(n):
    """
    Get a viridis color sequence.

    Parameters:
    n - Number of colors.
    
    Returns:
    List of n hex codes as strings.
    """
    
    return([mCm.to_hex(cm['viridis_r'].resampled(n).colors[v]) for v in range(n)])





def returnToOriginal() :
    '''

    Parameters :
    <none>

    Returns :
    EV - Expected Value of number of swaps.

    '''

    swaps = 0
    states = [1]+[0]*5

    while states != [1]*6 :

        if swaps % 2 == 0 :

            swaps += 1
            states[rC([3, 4, 5])] = 1

        elif swaps % 2 == 1 :

            swaps += 1
            states[rC([0, 1, 2])] = 1

    return(swaps)





def clearAx(ax):
    
    """
    Clear axes.

    Parameters:
    ax - Current ax.
    
    Returns:
    ax - Updated ax.
    """    
    
    ax.spines["top"].set_visible(False)
    ax.spines["right"].set_visible(False)
    ax.spines["bottom"].set_visible(False)
    ax.spines["left"].set_visible(False)
    ax.set_xticks([])
    ax.set_yticks([])
    return(ax)







def bipartiteGraph() :
    
    """
    Display the graph of the 6 states.

    Parameters:
    <none>.

    Returns:
    Plot.
    """

    fig = plt.figure(figsize = (10, 5))
    ax = fig.add_subplot(xlim = (-5, 5),
                         ylim = (-2, 2))
    fig.subplots_adjust(left = 0, bottom = 0, right = 1, top = 1)
    
    #Remove axes and ticks.
    ax = clearAx(ax)

    colors = getViridis(7)
    
    text = [("A, B, C", (-1.5, 1.5), colors[1], 'right'),
            ("B, C, A", (-1.5, 0.0), colors[1], 'right'),
            ("C, A, B", (-1.5, -1.5), colors[1], 'right'),

            ("A, C, B", (1.5, 1.5), colors[4], 'left'),
            ("C, B, A", (1.5, 0.0), colors[4], 'left'),
            ("B, A, C", (1.5, -1.5), colors[4], 'left')]

    for tt in text :
        ax.annotate(text = tt[0],
                    xy = tt[1],
                    c = tt[2],
                    size = 64,
                    ha = tt[3],
                    va = 'center',
                    zorder = 4)


    for a in [0, 1, 2] :
        for b in [3, 4, 5] :

            ax.annotate("",
                        xytext = text[a][1],
                        xy = text[b][1],
                        arrowprops = dict(arrowstyle = "<->, head_length = 2, head_width = 0.5",
                                          linewidth = 3,
                                          color = colors[6]))
        
    fig.savefig("2026.09.18 EC.png", bbox_inches = "tight")
