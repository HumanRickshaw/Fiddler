def maxRangeN(a, t, p) :
    '''
    
    Parameters :
        a - probability of getting either end range value.
        t - threshold of probability of getting range.
        p - print option, output the probability of getting range.

    Returns :
        N - Number of rolls needed to get threshold.

    '''

    N = 1
    
    rolls = [1-2*a, a, a, 0]
    if p :
        print(N, rolls[3])
    while rolls[3] <= t :
        

        newrolls = [rolls[0] * (1-2*a),
                    rolls[0] * a + rolls[1] * (1-a),
                    rolls[0] * a + rolls[2] * (1-a),
                    rolls[1] * a + rolls[2] * a + rolls[3]]

        N += 1
        rolls = newrolls
        if p :
            print(N, rolls[3])
        
    return(N)
