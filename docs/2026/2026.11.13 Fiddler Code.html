def clearAx(ax):
    
    """
    Clear axes.

    Parameters:
    ax - Current ax.
    
    Returns:
    ax - Updated ax.
    """    
    
    ax.spines["top"].set_visible(False)
    ax.spines["right"].set_visible(False)
    ax.spines["bottom"].set_visible(False)
    ax.spines["left"].set_visible(False)
    ax.set_xticks([])
    ax.set_yticks([])
    return(ax)


def distanceToRed(x1, y1, x2, y2, t) :

    """
    Calculate the perpendicular distance of a path to the red ball.

    Parameters:
        (x1, y1) & (x2, y2) - Two points define line.

    Returns :
        D - Distance from line to (0, 5)
        (x,y) - point of intersection
    """

    m = (y2 - y1) / (x2 - x1)

    #y-y1 = m(x - x1)
    #y = mx + (-mx1 + y1)

    a = m
    b = -1
    c = -m*x1 + y1

    n = a*(0) + b*(5) + c
    d = mS(a**2 + b**2)
    D = abs(n) / d

    #Base of Perpendicular
    x = 0 - a*n/(d**2)
    y = 5 - b*n/(d**2)

    #Point that is exactly 2 away.
    A = m**2+1
    B = 2*x1 + 2*m*y1 - 10*m
    C = x1**2 + (y1-5)**2 - 4

    try :
        disc = mS(B**2 - 4*A*C)
        xr1 = (-B+disc)/(2*A) + x1
        xr2 = (-B-disc)/(2*A) + x1
        yr1 = m*(xr1-x1) + y1
        yr2 = m*(xr2-x1) + y1

        if t < 44.245 :
            if yr1 < yr2 :
                xy = (xr1, yr1)
            else :
                xy = (xr2, yr2)

        elif t < 44.661 :
            xy = (xr2, yr2)

        elif t < 66.422 :
            xy = (xr1, yr1)

        else :
            xy = (xr2, yr2)
        
        return([D, xy])

    except :
        return([D, (x2, y2)])

def reflectionPoints(t) :

    #Radians.
    T = t*pi/180
    
    #0.1 ≤ θ < 44.3
    side_to_side = [[0] + [4, -4]*215,
                    [0] + [(4+8*y)*tan(T) for y in range(430)]]
    #44.3 ≤ θ < 66.0
    around_the_table = [[0, 4, 8-9/tan(T), -4, 4, -4],
                        [0, 4*tan(T), 9, 18-12*tan(T), 18-20*tan(T), 18-28*tan(T)]]
    
    #66.0 ≤ θ ≤ 90.0
    freak = [[0, 9/tan(T), 4, -4],
             [0, 9, 18-4*tan(T), 18-12*tan(T)]]

    if t < 44.246 :
        points = side_to_side

    elif t < 65.921 :
        points = around_the_table

    else :
        points = freak

    Xs = []
    Ys = []
    for i in range(len(points[0])) :
        x1 = points[0][i]
        y1 = points[1][i]
        Xs.append(x1)
        Ys.append(y1)

        if i == len(points[0])-1 :
            return([Xs, Ys])

        else :

            #If distance is <= 2, then stop at that point.
            D, xy = distanceToRed(x1, y1, points[0][i+1], points[1][i+1], t)

            if D <= 2 :
                Xs.append(xy[0])
                Ys.append(xy[1])
                return([Xs, Ys])
