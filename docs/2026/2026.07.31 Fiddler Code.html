def clearAx(ax):
    
    """
    Clear axes.

    Parameters:
    ax - Current ax.
    
    Returns:
    ax - Updated ax.
    """    
    
    ax.spines["top"].set_visible(False)
    ax.spines["right"].set_visible(False)
    ax.spines["bottom"].set_visible(False)
    ax.spines["left"].set_visible(False)
    ax.set_xticks([])
    ax.set_yticks([])
    return(ax)



def helixPath() :
    
    """
    Draw a single parallelogram on 4 stars of the flag array.

    Parameters:
    i : Parallelogram index.

    Returns:
    Plot.
    """

    sns.set()
    fig = plt.figure(figsize = (10, 5))
    ax = fig.add_subplot(xlim = (-5, 5),
                         ylim = (-2, 2))
    fig.subplots_adjust(left = 0, bottom = 0, right = 1, top = 1)
    
    #Remove axes and ticks.
    ax = clearAx(ax)
    
    #Background.
    ax.add_patch(mpR(xy = (-4, -2),
                     width = 8,
                     height = 4,
                     edgecolor = c[0],
                     facecolor = c[0],
                     zorder = 1))
    
    for r in [[(-4, 0.9), (1, 1), '-', c[1], 4],
              [(2.1, 4), (1, 1), '-', c[1], 4],
              [(-4, -0.6), (0, 0), '-', c[1], 4],
              [(0.6, 4), (0, 0), '-', c[1], 4],
              [(-4, -3.1), (-1, -1), '-', c[1], 4],
              [(-3.1, -0.1), (-1, -1), ':', c[1], 4],
              [(0.1, 4), (-1, -1), '-', c[1], 4],
             
              [(-4, 4), (-7/5, 91/50), ':', c[2], 1.5],
              [(-3.5, 2.5), (-2, 2), ':', c[2],  1.5],
              [(-2.5, 3.5), (-2, 2), ':', c[2], 1.5],
              [(-0.5, 1.5), (-2, 2), ':', c[2], 1.5],] :
        ax.plot(r[0], r[1], ls = r[2], c = r[3], lw = r[4], zorder = 2)
       
    #Background.
    ax.add_patch(mpR(xy = (-5, -2),
                     width = 1,
                     height = 4,
                     edgecolor = 'w',
                     facecolor = 'w',
                     zorder = 3))
    ax.add_patch(mpR(xy = (4, -2),
                     width = 1,
                     height = 4,
                     edgecolor = 'w',
                     facecolor = 'w',
                     zorder = 3))
        
    text = [('Ring A', (-3.7, 1.2), c[1], 24, 'left'),

            ('Ring C', (-3.7, -0.8), c[1], 24, 'left'),
